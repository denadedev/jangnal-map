```ts
import { readFile } from "node:fs/promises";
import { parse } from "csv-parse/sync";
import type { RawMarket } from "./types.js";

const required = ["시장명", "시장유형", "시장개설주기", "위도", "경도"];

export async function readMarketsCsv(path: string): Promise<RawMarket[]> {
  const text = await readFile(path, "utf8");
  const rows = parse(text, { columns: true, bom: true, skip_empty_lines: true }) as RawMarket[];
  const first = rows[0] as Record<string, unknown> | undefined;
  if (!first || required.some((key) => !(key in first))) {
    throw new Error("전통시장 CSV 필수 컬럼이 없습니다");
  }
  return rows;
}
```

```ts
import type { NormalizedMarket, RawMarket } from "./types.js";

const nullableText = (value: string): string | null => value.trim() || null;
const nullableNumber = (value: string): number | null => {
  if (!value.trim()) return null;
  const parsed = Number(value.replaceAll(",", ""));
  return Number.isFinite(parsed) ? parsed : null;
};

export function normalizeMarket(raw: RawMarket): NormalizedMarket {
  const parking = raw.주차장보유여부.trim().toUpperCase();
  return {
    name: raw.시장명.trim(),
    marketType: raw.시장유형.trim(),
    roadAddress: nullableText(raw.소재지도로명주소),
    lotAddress: nullableText(raw.소재지지번주소),
    scheduleRaw: nullableText(raw.시장개설주기),
    latitude: nullableNumber(raw.위도),
    longitude: nullableNumber(raw.경도),
    storeCount: nullableNumber(raw.점포수),
    products: raw.취급품목.split("+").map((item) => item.trim()).filter(Boolean),
    hasParking: parking === "Y" ? true : parking === "N" ? false : null,
    phone: nullableText(raw.전화번호),
    referenceDate: nullableText(raw.데이터기준일자),
    status: "운영"
  };
}
```

- [ ] **Step 3: 공통적인 숫자 쌍만 최소 구현한다**

```ts
export type ParsedSchedule =
  | { kind: "daily" }
  | { kind: "digit-pair"; days: [number, number] }
  | { kind: "weekday"; raw: string }
  | { kind: "unknown"; raw: string };

export function parseSchedule(input: string | null): ParsedSchedule {
  const raw = input?.trim() ?? "";
  if (raw === "매일") return { kind: "daily" };

  const pair = raw.match(/(?:매월\s*)?(\d{1,2})일?\s*[,·/]\s*(\d{1,2})일?(?:장)?/);
  if (pair) {
    const first = Number(pair[1]);
    const second = Number(pair[2]);
    const valid = (day: number) => day === 0 || (day >= 1 && day <= 31);
    if (valid(first) && valid(second)) {
      return { kind: "digit-pair", days: [first, second] };
    }
  }

  if (/요일|째/.test(raw)) return { kind: "weekday", raw };
  return { kind: "unknown", raw };
}
```

- [ ] **Step 4: 숫자 범위 검증 테스트를 추가한다**

`0`은 매월 10·20·30일을 의미할 수 있으므로 허용하되, 다른 숫자는 `1..31` 범위만 허용한다. `32·37일장`은 `unknown`이어야 한다.

- 중복 후보 키: 공백을 제거한 `시장명 + 도로명주소 또는 지번주소`
- 알 수 없는 장날: 원문별 개수를 내림차순으로 집계

```ts
import type { NormalizedMarket } from "./types.js";
import { parseSchedule } from "./parse-schedule.js";

export interface AuditResult {
  totalRows: number;
  periodicCandidates: number;
  validCoordinates: number;
  parseableSchedules: number;
  fieldCompleteness: {
    address: number;
    phone: number;
    parking: number;
    referenceDate: number;
  };
  unknownScheduleCounts: Array<{ raw: string; count: number }>;
  duplicateCandidates: Array<{ key: string; names: string[] }>;
  publishableRows: number;
}

const hasValidCoordinates = (market: NormalizedMarket): boolean =>
  market.latitude !== null &&
  market.longitude !== null &&
  market.latitude >= 33 &&
  market.latitude <= 39.5 &&
  market.longitude >= 124 &&
  market.longitude <= 132;

const isPeriodicCandidate = (market: NormalizedMarket): boolean =>
  market.marketType.includes("정기") ||
  (market.scheduleRaw !== null && parseSchedule(market.scheduleRaw).kind !== "daily");

export function analyzeMarkets(markets: NormalizedMarket[]): AuditResult {
  const candidates = markets.filter(isPeriodicCandidate);
  const parsed = candidates.map((market) => ({ market, schedule: parseSchedule(market.scheduleRaw) }));

  const unknown = new Map<string, number>();
  for (const item of parsed) {
    if (item.schedule.kind === "unknown") {
      unknown.set(item.schedule.raw, (unknown.get(item.schedule.raw) ?? 0) + 1);
    }
  }

  const duplicateMap = new Map<string, string[]>();
  for (const market of markets) {
    const address = market.roadAddress ?? market.lotAddress ?? "";
    const key = `${market.name}${address}`.replaceAll(/\s/g, "");
    const names = duplicateMap.get(key) ?? [];
    names.push(market.name);
    duplicateMap.set(key, names);
  }

  return {
    totalRows: markets.length,
    periodicCandidates: candidates.length,
    validCoordinates: candidates.filter(hasValidCoordinates).length,
    parseableSchedules: parsed.filter(({ schedule }) => schedule.kind === "digit-pair").length,
    fieldCompleteness: {
      address: markets.filter((market) => market.roadAddress !== null || market.lotAddress !== null).length,
      phone: markets.filter((market) => market.phone !== null).length,
      parking: markets.filter((market) => market.hasParking !== null).length,
      referenceDate: markets.filter((market) => market.referenceDate !== null).length
    },
    unknownScheduleCounts: [...unknown.entries()]
      .map(([raw, count]) => ({ raw, count }))
      .sort((a, b) => b.count - a.count),
    duplicateCandidates: [...duplicateMap.entries()]
      .filter(([, names]) => names.length > 1)
      .map(([key, names]) => ({ key, names })),
    publishableRows: parsed.filter(
      ({ market, schedule }) => schedule.kind === "digit-pair" && hasValidCoordinates(market)
    ).length
  };
}
```

```ts
import { mkdir, writeFile } from "node:fs/promises";
import { dirname } from "node:path";
import type { AuditResult } from "./analyze.js";
import type { NormalizedMarket } from "./types.js";
import { parseSchedule } from "./parse-schedule.js";

const percent = (value: number, total: number): string =>
  total === 0 ? "0.0%" : `${((value / total) * 100).toFixed(1)}%`;

const verdict = (result: AuditResult): string => {
  const ratio = result.periodicCandidates === 0 ? 0 : result.publishableRows / result.periodicCandidates;
  if (ratio >= 0.9) return "사용 가능";
  if (ratio >= 0.7) return "보완 후 사용 가능";
  return "사용 불가";
};

const selectRegionalSample = (markets: NormalizedMarket[]): NormalizedMarket[] => {
  const groups = new Map<string, NormalizedMarket[]>();
  for (const market of markets) {
    const schedule = parseSchedule(market.scheduleRaw);
    if (schedule.kind !== "digit-pair" || market.latitude === null || market.longitude === null) continue;
    const region = (market.roadAddress ?? market.lotAddress ?? "주소없음").split(" ")[0] ?? "주소없음";
    groups.set(region, [...(groups.get(region) ?? []), market]);
  }
  const sample: NormalizedMarket[] = [];
  while (sample.length < 100 && [...groups.values()].some((rows) => rows.length > 0)) {
    for (const rows of groups.values()) {
      const market = rows.shift();
      if (market) sample.push(market);
      if (sample.length === 100) break;
    }
  }
  return sample;
};

export async function writeAuditOutputs(
  markets: NormalizedMarket[],
  result: AuditResult,
  reportPath: string,
  samplePath: string
): Promise<void> {
  const report = `# 전국전통시장 데이터 조사 결과

## 결론
- ${verdict(result)}

## 전체 현황
- 전체 행 수: ${result.totalRows}
- 정기시장 후보 수: ${result.periodicCandidates}
- 바로 게시 가능한 시장: ${result.publishableRows} (${percent(result.publishableRows, result.periodicCandidates)})

## 데이터 품질
- 좌표 유효: ${result.validCoordinates} (${percent(result.validCoordinates, result.periodicCandidates)})
- 장날 파싱 가능: ${result.parseableSchedules} (${percent(result.parseableSchedules, result.periodicCandidates)})
- 중복 후보: ${result.duplicateCandidates.length}
- 주소 있음: ${result.fieldCompleteness.address} (${percent(result.fieldCompleteness.address, result.totalRows)})
- 전화번호 있음: ${result.fieldCompleteness.phone} (${percent(result.fieldCompleteness.phone, result.totalRows)})
- 주차정보 있음: ${result.fieldCompleteness.parking} (${percent(result.fieldCompleteness.parking, result.totalRows)})

## 파싱 불가능한 장날 원문
${result.unknownScheduleCounts.slice(0, 50).map(({ raw, count }) => `- ${raw || "(빈 값)"}: ${count}`).join("\n")}

## 수동 보완 예상량
- 장날 또는 좌표 보완 대상: ${result.periodicCandidates - result.publishableRows}
- 중복 확인 대상: ${result.duplicateCandidates.length}

## 서비스 진행 판정
- ${verdict(result)}
`;

  const sample = selectRegionalSample(markets).map((market) => ({
    name: market.name,
    latitude: market.latitude,
    longitude: market.longitude,
    scheduleRaw: market.scheduleRaw,
    schedule: parseSchedule(market.scheduleRaw),
    referenceDate: market.referenceDate,
    status: market.status
  }));

  await mkdir(dirname(reportPath), { recursive: true });
  await mkdir(dirname(samplePath), { recursive: true });
  await writeFile(reportPath, report, "utf8");
  await writeFile(samplePath, `${JSON.stringify(sample, null, 2)}\n`, "utf8");
}
```
